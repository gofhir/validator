// valor (la cadena SliceOf y la definición propia). Una constraint con la misma clave y texto distinto en el hijo del

func TestReviewR2LayersChildrenAreWalkedBeforeTheOtherDefinitions(t *testing.T) {
	const want = "error @ Patient.identifier[0].value: Constraint failed: gk: 'gk text' (defined in http://example.org/hm/tb-10b)"

	})
}
