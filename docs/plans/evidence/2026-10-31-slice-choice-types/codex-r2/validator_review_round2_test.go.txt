package validator

import (
	"context"
	"strings"
	"testing"

	"github.com/gofhir/validator/v2/pkg/issue"
)

func TestReviewOwnNestedChoice(t *testing.T) {
	const def = `{"resourceType":"StructureDefinition","url":"http://example.org/timing-ext","name":"TimingExt","status":"active","kind":"complex-type","abstract":false,"type":"Extension","baseDefinition":"http://hl7.org/fhir/StructureDefinition/Extension","derivation":"constraint","context":[{"type":"element","expression":"Patient"}],"differential":{"element":[{"id":"Extension","path":"Extension"},{"id":"Extension.extension","path":"Extension.extension","max":"0"},{"id":"Extension.url","path":"Extension.url","fixedUri":"http://example.org/timing-ext"},{"id":"Extension.value[x]","path":"Extension.value[x]","type":[{"code":"Timing"}]},{"id":"Extension.value[x].repeat.bounds[x]","path":"Extension.value[x].repeat.bounds[x]","type":[{"code":"Period"}]}]}}`
	const data = `{"resourceType":"Patient","extension":[{"url":"http://example.org/timing-ext","valueTiming":{"repeat":{"boundsDuration":{"value":1,"unit":"d","system":"http://unitsofmeasure.org","code":"d"}}}}]}`
	for _, version := range orderVersions {
		t.Run(version, func(t *testing.T) {
			v := orderValidator(t, version, def)
			res, err := v.Validate(context.Background(), []byte(data))
			if err != nil {
				t.Fatal(err)
			}
			want := at{id: issue.DiagStructureTypeNotAllowed, expr: "Patient.extension[0].valueTiming.repeat.boundsDuration"}
			if !want.in(res) {
				t.Fatalf("missing nested choice type error; issues: %+v", res.Issues)
			}
		})
	}
}

func TestReviewDeclaredAbsoluteChild(t *testing.T) {
	defs := extValueDefs(t)
	for i, d := range defs {
		if strings.Contains(d, `"name": "BoolExt"`) {
			defs[i] = strings.ReplaceAll(d, `"expression": "Patient"`, `"expression": "Element"`)
		}
		if strings.Contains(d, `"name": "NestedExt"`) {
			d = strings.ReplaceAll(d, `"fixedUri": "part"`, `"fixedUri": "http://example.org/StructureDefinition/bool-ext"`)
			// Se omite la restricción del valor en el padre: la define bool-ext.
			start := strings.Index(d, "   {\n    \"id\": \"Extension.extension:part.value[x]\"")
			end := strings.Index(d[start:], "   {\n    \"id\": \"Extension.url\"") + start
			d = d[:start] + d[end:]
			defs[i] = d
		}
	}
	const data = `{"resourceType":"Patient","extension":[{"url":"http://example.org/StructureDefinition/nested-ext","extension":[{"url":"http://example.org/StructureDefinition/bool-ext","valueString":"bad"}]}]}`
	for _, version := range orderVersions {
		t.Run(version, func(t *testing.T) {
			v := orderValidator(t, version, defs...)
			res, err := v.Validate(context.Background(), []byte(data))
			if err != nil {
				t.Fatal(err)
			}
			want := at{id: issue.DiagStructureTypeNotAllowed, expr: "Patient.extension[0].extension[0].valueString", text: "'http://example.org/StructureDefinition/bool-ext'"}
			if !want.in(res) {
				t.Fatalf("missing absolute child type error; issues: %+v", res.Issues)
			}
		})
	}
}
