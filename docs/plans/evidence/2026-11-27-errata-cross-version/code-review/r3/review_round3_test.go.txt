func TestErrataCrossVersionReviewRound2MixedBrokenAncestry(t *testing.T) {
 for _, ending := range []string{"missing", "cycle"} {
  t.Run(ending, func(t *testing.T) {
   r := ecvRegistry(t, "4.0.1")
   base, parent := "http://example.org/r2/r4-base", "http://example.org/r2/r5-parent"
   end := "http://example.org/r2/missing"
   if ending == "cycle" { end = base }
   var published map[string]json.RawMessage
   if err := json.Unmarshal(ecvOfficialSDs(t,"4.0.1")[ecvCore+"Bundle"], &published); err != nil {t.Fatal(err)}
   published["url"] = json.RawMessage(ecvJSON(base))
   published["derivation"] = json.RawMessage(`"constraint"`)
   published["baseDefinition"] = json.RawMessage(ecvJSON(parent))
   delete(published,"differential")
   ecvLoad(t,r,ecvSD(parent,"5.0.0","resource","Bundle",end,`{"id":"Bundle","path":"Bundle"}`),ecvJSON(published),ecvSD(base+"-leaf","4.0.1","resource","Bundle",base,`{"id":"Bundle","path":"Bundle","short":"leaf"}`))
   if !constraintKeys(ecvElement(t,r.GetByURL(base),"Bundle"))["bdl-15"] {t.Fatal("base lacks bdl-15")}
   sd := ecvSnapshot(t,r,base+"-leaf")
   if !constraintKeys(ecvElement(t,sd,"Bundle"))["bdl-15"] {t.Errorf("R4 published base has bdl-15 but leaf loses it")}
  })
 }
}

func TestErrataCrossVersionReviewRound2NonStringMember(t *testing.T) {
 for _, value := range []string{`123`,`true`,`{}`,`[]`,`null`} {
  t.Run(value,func(t *testing.T){
   raw := `{"element":[{"id":"Reference","path":"Reference","constraint":[`+ecvConstraintJSON("ref-1", ecvExpression(t,"4.0.1","Reference","ref-1"))+`,{"key":"own","expression":`+value+`}]}]}`
   var s Snapshot
   err := json.Unmarshal([]byte(raw), &s)
   if value != "null" {if err == nil {t.Fatal("non-string accepted")}; t.Logf("rejected: %v",err);return}
   if err != nil {t.Fatal(err)}
   correctElements("4.0.1",s.Element)
   c := ecvRawConstraint(t,&s.Element[0],"own")
   if string(c["expression"]) != "null" {t.Errorf("null changed: %s",c["expression"])}
  })
 }
}

func TestErrataCrossVersionReviewRound3RemovedReplaced(t *testing.T) {
 for _, replacement := range []string{"", "false"} {
  r := ecvRegistry(t,"5.0.0")
  var published map[string]any
  json.Unmarshal(ecvOfficialSDs(t,"5.0.0")[ecvCore+"Bundle"], &published)
  published["url"]="http://example.org/r3/base"
  delete(published,"differential")
  elements:=published["snapshot"].(map[string]any)["element"].([]any)
  root:=elements[0].(map[string]any)
  cs:=root["constraint"].([]any)
  kept:=[]any{}
  for _, c:=range cs { m:=c.(map[string]any); if m["key"]=="bdl-15" {if replacement=="" {continue};m["expression"]=replacement};kept=append(kept,c) }
  root["constraint"]=kept
  ecvLoad(t,r,ecvJSON(published),ecvBundleDefinition("http://example.org/r3/leaf","4.0.1","http://example.org/r3/base",""))
  e:=ecvElement(t,ecvSnapshot(t,r,"http://example.org/r3/leaf"),"Bundle")
  if replacement=="" {if constraintKeys(e)["bdl-15"] {t.Fatal("resurrected removed invariant")}} else if ecvConstraintOf(t,e,"bdl-15").Expression!=replacement {t.Fatal("replaced invariant overwritten")}
 }
}

}

// These probes establish the limits of generic inheritance. They use synthetic
// errata because the current addedConstraints table only targets Bundle's root.
func TestErrataCrossVersionReviewRound3SyntheticPaths(t *testing.T) {
 saved:=addedConstraints
 defer func(){addedConstraints=saved}()
 synthetic:=Constraint{Key:"r3-added",Severity:"error",Human:"probe",Expression:"false"}
 addedConstraints=append(slices.Clone(saved),addedConstraint{fhirVersion:fhirR4,path:"Observation.effective[x]",constraint:synthetic},addedConstraint{fhirVersion:fhirR4,path:"Timing.repeat",constraint:synthetic})
 r:=ecvRegistry(t,"4.0.1")
 // sharedVersion was loaded before the synthetic table: apply only to private copies.
 for _,name:=range []string{"Observation","Timing"} {
  var p map[string]any;json.Unmarshal(ecvOfficialSDs(t,"4.0.1")[ecvCore+name],&p)
  p["url"]="http://example.org/r3/"+name
  delete(p,"differential")
  ecvLoad(t,r,ecvJSON(p))
 }
 ecvLoad(t,r,
  ecvSD("http://example.org/r3/renamed","5.0.0","resource","Observation","http://example.org/r3/Observation",`{"id":"Observation.effectiveDateTime","path":"Observation.effectiveDateTime","type":[{"code":"dateTime"}]}`),
  ecvSD("http://example.org/r3/unrolled","5.0.0","complex-type","Dosage",ecvCore+"Dosage",`{"id":"Dosage.timing","path":"Dosage.timing","type":[{"code":"Timing","profile":["http://example.org/r3/Timing"]}]}`,`{"id":"Dosage.timing.repeat","path":"Dosage.timing.repeat","min":1}`))
 renamed:=ecvSnapshot(t,r,"http://example.org/r3/renamed")
 for _,e:=range renamed.Snapshot.Element {if e.ID=="Observation.effective[x]:effectiveDateTime" && !constraintKeys(&e)[synthetic.Key] {t.Fatal("renamed choice lost synthetic invariant")}}
 ecvLoad(t,r,
  `{"resourceType":"StructureDefinition","url":"http://example.org/r3/crbase","name":"Host","status":"active","fhirVersion":"5.0.0","kind":"logical","abstract":false,"type":"Host","derivation":"specialization","snapshot":{"element":[{"id":"Host","path":"Host"},{"id":"Host.part","path":"Host.part","contentReference":"http://example.org/r3/Timing#Timing"}]}}`,
  ecvSD("http://example.org/r3/cr","5.0.0","logical","Host","http://example.org/r3/crbase",`{"id":"Host.part.repeat","path":"Host.part.repeat","min":1}`))
 cr:=ecvElement(t,ecvSnapshot(t,r,"http://example.org/r3/cr"),"Host.part.repeat")
 t.Logf("synthetic external contentReference invariant present=%v",constraintKeys(cr)[synthetic.Key])
 e:=ecvElement(t,ecvSnapshot(t,r,"http://example.org/r3/unrolled"),"Dosage.timing.repeat")
 t.Logf("synthetic unrolled invariant present=%v (no counterpart in Dosage base)",constraintKeys(e)[synthetic.Key])
 // The stripped fallback jumps over a sliced parent, so an invariant on a:b
 // cannot be inherited by a:b/c when absent from the unsliced a.
 base:=&Snapshot{Element:[]ElementDefinition{{ID:"Patient.identifier"},{ID:"Patient.identifier:b",Constraint:[]Constraint{synthetic}}}}
 derived:=&Snapshot{Element:[]ElementDefinition{{ID:"Patient.identifier:b/c"}}}
 inheritAddedConstraints(derived,base)
 t.Logf("synthetic reslice invariant present=%v",constraintKeys(&derived.Element[0])[synthetic.Key])
}

func TestErrataCrossVersionReviewRound3MutationDomain(t *testing.T) {
 r:=ecvRegistry(t,"4.0.1")
 for _,own:=range []string{"","false"} {
  url:="http://example.org/r3/mutation-"+own
  ecvLoad(t,r,ecvBundleDefinition(url,"5.0.0",ecvCore+"Bundle",own))
  sd:=r.GetByURL(url)
  base:=r.GetByURL(ecvCore+"Bundle").Snapshot
  choices,err:=r.choiceIndexFor(context.Background(),"5.0.0");if err!=nil {t.Fatal(err)}
  pre,_,err:=r.applyDifferential(context.WithValue(context.Background(),profileVersionKey{},"5.0.0"),base,sd.Differential,choices);if err!=nil {t.Fatal(err)}
  for i,b:=range base.Element {
   e:=&pre.Element[i]
   for _,c:=range b.Constraint {
    missing:=!constraintKeys(e)[c.Key]
    if missing && !isAddedConstraint(c) {t.Fatalf("mutation20 differs: %s %s",e.ID,c.Key)}
    if missing {t.Logf("own=%q: missing %s/%s expr=%q",own,e.ID,c.Key,c.Expression)}
   }
  }
  // Compare concrete outputs for the two mutant predicates, including raw,
  // before source stamping, independently of whether tests detect a difference.
  all,_:=deepCopyElements(pre.Element);key,_:=deepCopyElements(pre.Element)
  for i,b:=range base.Element {
   for _,c:=range b.Constraint {
    if !constraintKeys(&all[i])[c.Key] {all[i].Constraint=append(all[i].Constraint,c)}
    if c.Key=="bdl-15" && !constraintKeys(&key[i])[c.Key] {key[i].Constraint=append(key[i].Constraint,c)}
   }
  }
  inheritAddedConstraints(pre,base)
  want:=ecvJSON(pre.Element)
  if ecvJSON(all)!=want || ecvJSON(key)!=want {t.Fatal("mutant output differs")}
  t.Logf("own=%q: production, mutation20 and mutation22 outputs equal",own)
 }
}
