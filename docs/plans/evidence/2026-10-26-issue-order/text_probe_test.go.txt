package validator

import (
	"context"
	"fmt"
	"os"
	"strings"
	"testing"
)

func TestZZTextProbe(t *testing.T) {
	v, err := New(WithVersion("4.0.1"))
	if err != nil {
		t.Fatal(err)
	}
	cases := map[string]string{
		"choice": `{"resourceType":"Observation","status":"final","code":{"text":"x"},"valueString":"a","valueInteger":1,"valueBoolean":true,"effectiveDateTime":"2020","effectivePeriod":{"start":"2020"}}`,
		"ext":    `{"resourceType":"Patient","extension":[{"url":"http://hl7.org/fhir/StructureDefinition/patient-birthPlace","valueString":"a","valueAddress":{"city":"x"}}]}`,
	}
	out, _ := os.Create(os.Getenv("PROBE_OUT"))
	defer out.Close()
	for name, c := range cases {
		seen := map[string]int{}
		for range 40 {
			r, _ := v.Validate(context.Background(), []byte(c))
			var l []string
			for _, is := range r.Issues {
				l = append(l, is.MessageID+"|"+is.Diagnostics+"|"+strings.Join(is.Expression, ","))
			}
			seen[strings.Join(l, "\n")]++
		}
		fmt.Fprintf(out, "== %s: %d distinct outputs\n", name, len(seen))
		for k := range seen {
			fmt.Fprintf(out, "%s\n--\n", k)
		}
	}
}
